import { vi } from 'vitest'

// Factory function — chame no beforeEach de cada arquivo de teste para
// garantir vi.fn() frescos e isolamento completo entre testes.
//
// Uso:
//   let mocks
//   beforeEach(() => { mocks = createFirebaseMocks() })
//   vi.mock('../firebase', () => createFirebaseMocks())
//
// PITFALL: onAuthStateChanged deve chamar cb(null) SINCRONAMENTE
// e retornar () => {} — o AuthProvider chama o retorno no unmount.
export function createFirebaseMocks() {
  return {
    // Instâncias dos serviços (objetos vazios — nunca usam a rede)
    db: {},
    storage: {},
    auth: {},

    // Auth — mock síncrono crítico: chama cb(null) imediatamente
    // Retorna unsubscribe como () => {} para evitar TypeError no cleanup
    onAuthStateChanged: vi.fn((auth, cb) => {
      cb(null)
      return () => {}
    }),
    signInWithEmailAndPassword: vi.fn(),
    createUserWithEmailAndPassword: vi.fn(),
    signOut: vi.fn(),

    // Firestore — operações de leitura
    getDocs: vi.fn(),
    getDoc: vi.fn(),
    getCountFromServer: vi.fn(),

    // Firestore — operações de escrita
    addDoc: vi.fn(),
    updateDoc: vi.fn(),
    setDoc: vi.fn(),
    deleteDoc: vi.fn(),

    // Firestore — helpers de query e referência
    collection: vi.fn(),
    doc: vi.fn(),
    query: vi.fn(),
    orderBy: vi.fn(),
    where: vi.fn(),
    limit: vi.fn(),

    // Firestore — server timestamp
    // Retorna objeto com toDate() para não quebrar fmtDate/fmtDatetime
    serverTimestamp: vi.fn(() => ({ toDate: () => new Date('2024-01-15') })),

    // Storage
    storageRef: vi.fn(),
    uploadBytes: vi.fn(),
    getDownloadURL: vi.fn(),

    // Constantes e utilitários
    SUPERADMIN_EMAIL: 'superadmin@test.com',
    PLANOS: {
      basico:     { limiteOS: 50,  preco: 97  },
      pro:        { limiteOS: -1,  preco: 197 },

    },
    refChecklist: vi.fn(),
    refConfig: vi.fn(),
  }
}

Criar mp-react/src/test-utils/renderWithProviders.jsx com o conteúdo EXATO abaixo (D-11):

```jsx
import { render } from '@testing-library/react'
import { MemoryRouter } from 'react-router-dom'
import { AuthContext } from '../contexts/AuthContext'

// FakeAuthProvider — fornece contexto de auth falso sem conectar ao Firebase.
// loadingAuth: false é OBRIGATÓRIO — componentes renderizam null com true.
//
// Uso: renderWithProviders(<MeuComponente />, { authOverrides: { usuario: mockUser } })
function FakeAuthProvider({ children, authValue }) {
  return (
    <AuthContext.Provider value={authValue}>
      {children}
    </AuthContext.Provider>
  )
}

// Valor padrão do auth — representa usuário deslogado.
// Todos os campos que AuthContext.Provider fornece devem estar presentes.
const DEFAULT_AUTH = {
  usuario: null,
  empresaId: null,
  isSuperAdmin: false,
  loadingAuth: false,
  login: () => Promise.resolve(),
  logout: () => Promise.resolve(),
  cadastrar: () => Promise.resolve(),
}

// renderWithProviders — substituto do render() do RTL para componentes
// que dependem de React Router e/ou AuthContext.
//
// Opções:
//   route    — string com a rota inicial (default: '/')
//   authOverrides — objeto com campos a sobrescrever no DEFAULT_AUTH
//
// Exemplo:
//   const { getByText } = renderWithProviders(<MinhaPage />, {
//     route: '/empresa-teste/admin',
//     authOverrides: { usuario: { uid: '123', email: 'admin@test.com' }, empresaId: '123' }
//   })
export function renderWithProviders(ui, { route = '/', authOverrides = {} } = {}) {
  const authValue = { ...DEFAULT_AUTH, ...authOverrides }

  return render(
    <MemoryRouter initialEntries={[route]}>
      <FakeAuthProvider authValue={authValue}>
        {ui}
      </FakeAuthProvider>
    </MemoryRouter>
  )
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| FakeAuthProvider → componentes | Contexto de auth falso — isola completamente do Firebase Auth real |
| createFirebaseMocks() → testes | vi.fn() frescos por chamada — nenhum estado vaza entre testes |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-04 | Spoofing | FakeAuthProvider | accept | Por design — contexto falso para testes. Nunca expor em produção. FakeAuthProvider só existe em src/test-utils/ |
| T-01-05 | Tampering | createFirebaseMocks | mitigate | Factory retorna objetos novos em cada chamada (nunca compartilhados). clearMocks: true zera call counts. Usar beforeEach para recriar mocks |
| T-01-06 | Elevation of Privilege | DEFAULT_AUTH | mitigate | DEFAULT_AUTH tem isSuperAdmin: false explícito. Testes que testam superadmin devem passar authOverrides: { isSuperAdmin: true } explicitamente |
</threat_model>
