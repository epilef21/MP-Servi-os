```js
receitaOS        = Σ osDoMes (mo_seguradora + valor_deslocamento + material_cobrado_seguradora)
receitaPart      = Σ particulares[mes_referencia===m].valor_recebido
receita          = receitaOS + receitaPart                      // = receitaTotal do DRE
totalDeducoes    = Σ deducoes[mes_referencia===m].valor
custoVariavel    = Σ osDoMes (valor_prestador + material_custo_real)
                 + Σ particulares[mes_referencia===m].valor_custo

totalDespesas    = Σ despesas[mes_referencia===m].valor          // por mes_referencia (série é DRE-like, não caixa)
lucroLiquido     = receita − totalDeducoes − custoVariavel + saldoFinanceiro − totalDespesas
margem           = receita > 0 ? (lucroLiquido / receita) * 100 : 0
```
